# The libc-test tests (libc-test.wasm, build.sh) that fail on ecmaOS today, one per line with why. ../libc-test.test.ts
# fails when a test not listed here fails, and when one listed here passes: take it off the list then.

# System V IPC is not in the kernel
functional.ipc_msg          msgget: ENOSYS
functional.ipc_sem          semget: ENOSYS
functional.ipc_shm          shmget: ENOSYS

# threads
functional.pthread_mutex_pi priority inheritance (PTHREAD_PRIO_INHERIT) is ENOSYS
functional.pthread_robust   robust mutexes are ENOSYS; times out
regression.pthread-robust-detach  robust mutexes are ENOSYS
regression.pthread_create-oom     times out
regression.raise-race       pthread_kill is EINVAL; times out

# formatting and parsing
functional.strptime         %F, %s and %z are not parsed (musl has them only after 1.2.5)
functional.strtod           hexadecimal floats past 53 bits are not rounded to nearest
functional.strtold          WALI's musl formats no long double (SIGABRT); built with it, it misformats doubles
functional.wordexp          most cases pass; the rest are musl 1.2.3's wordexp and ecmash's eval (positional
                            parameters, quoting inside ${X-...}, syntax errors)

# processes and signals

regression.daemon-failure   fork: ENOSYS
regression.fflush-exit      fork: ENOSYS
regression.pthread_exit-dtor  fork: ENOSYS
regression.sigaltstack      sigaltstack: ENOSYS

# memory and limits
regression.malloc-brk-fail  times out
regression.malloc-oom       times out
regression.setenv-oom       times out
regression.flockfile-list   "0 is not an allocated pointer"
