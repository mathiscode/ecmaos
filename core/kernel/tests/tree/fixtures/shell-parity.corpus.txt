# Shell parity corpus.
#
# Each entry is `### group/name [flags]` followed by a script. Flags: `fs` runs the script in a fresh
# /tmp/pc directory. The golden output (stdout and exit status) comes from real bash 5.3 through
# `node generate-shell-parity.mjs`, so this is a conformance corpus: both the kernel shell and
# `/bin/ecmash` are checked against bash, not against each other. A shell that differs is listed in
# the `known` table of shell-parity.test.ts with its actual output and a reason, so a fix forces the
# entry to be removed from that table.
#
# Scripts must be deterministic and use only builtins and commands both shells have.

### quoting/single-double
x="a b"; echo "$x" '$x' $x
### quoting/escapes
echo "a\$b" 'it'"'"'s' "q\"q"
### quoting/word-splitting
x="a   b    c"; set -- $x; echo $#; set -- "$x"; echo $#
### expansion/defaults
x=hello; echo ${x:-d} ${y:-d} ${x:+alt} ${y:+alt} ${#x}

### expansion/command-subst

### expansion/command-subst-nested

### expansion/command-subst-status
